### 配置文件

创建 `config.js`（因为魔珐星云和大模型都不是免费的，这里设计为用户可以在页面输入自己的配置）：

```javascript
// 从localStorage读取配置
function getConfig(key, defaultValue) {
    try {
        const stored = localStorage.getItem(key);
        return stored !== null && stored !== "" ? stored : defaultValue;
    } catch (e) {
        return defaultValue;
    }
}

const xingyunConfig = {
    get appId() { return getConfig('xingyun_appId', ""); },
    get appSecret() { return getConfig('xingyun_appSecret', ""); },
    get gatewayServer() { return getConfig('xingyun_gatewayServer', 
        "https://nebula-agent.xingyun3d.com/user/v1/ttsa/session"); }
};

const doubaoConfig = {
    get apiKey() { return getConfig('doubao_apiKey', ""); },
    get apiUrl() { return getConfig('doubao_apiUrl', 
        "https://ark.cn-beijing.volces.com/api/v3/chat/completions"); },
    get model() { return getConfig('doubao_model', ""); }
};

// 保存配置
function saveConfig(config) {
    if (config.xingyun_appId !== undefined) 
        localStorage.setItem('xingyun_appId', config.xingyun_appId);
    if (config.xingyun_appSecret !== undefined) 
        localStorage.setItem('xingyun_appSecret', config.xingyun_appSecret);
    // ... 其他配置
}
```

> 用getter + localStorage，使项目可以支持代码配置，也支持页面配置，方便用户使用自己的Key。
>
> **配置信息保存在本地localStorage，不会上传到服务器**

### SDK初始化

初始化魔珐星云的SDK。

```javascript
let liteSDK = null;
let isInitialized = false;

async function initSDK() {
    if (isInitialized) return;

    showLoading(true);
    
    try {
        const config = xingyunConfig;
        
        // 创建XmovAvatar实例
        liteSDK = new XmovAvatar({
            containerId: '#avatar-active',
            appId: config.appId,
            appSecret: config.appSecret,
            gatewayServer: config.gatewayServer,

            // 事件回调
            onWidgetEvent(data) {
                console.log('SDK事件:', data);
                if (data.type === 'subtitle_on') {
                    updateSubtitle(data.text);
                }
            },

            onStateChange(state) {
                if (state === 'connected') {
                    onSDKConnected();
                }
            },

            onVoiceStateChange(status) {
                handleVoiceStateChange(status);
            },

            enableLogger: true
        });
        
        // 初始化SDK
        await liteSDK.init({
            onDownloadProgress: (progress) => {
                updateLoadingProgress(progress);
            }
        });

        isInitialized = true;
        hideLoading();
        
    } catch (error) {
        console.error('初始化失败:', error);
        alert('初始化失败: ' + error.message);
    }
}
```

### 语音播报

设置魔珐星云语音播放消息的方法。

```javascript
async function speakMessage(text) {
    if (!isInitialized || !liteSDK) {
        alert('SDK未初始化');
        return;
    }
    
    isSpeaking = true;
    
    try {
        // 调用speak方法
        const result = liteSDK.speak(text, true, true);
        
    } catch (error) {
        console.error('讲解失败:', error);
    }
}

// 停止讲解
async function stopSpeaking() {
    if (!isInitialized || !liteSDK) return;
    
    if (typeof liteSDK.offlineMode === 'function') {
        liteSDK.offlineMode();
    }
    
    isSpeaking = false;
    liteSDK.onlineMode();
}
```

### 流式调用API

调用的大模型获取智能讲解，使用流式输出，然后调用数字人的语音播报。

```javascript
async function callDoubaoAPI(prompt) {
    const config = doubaoConfig;

    if (!config.apiKey) {
        alert('请先配置API Key');
        openConfigModal();
        return;
    }

    const requestData = {
        model: config.model,
        stream: true,  // 开启流式输出
        messages: [
            { content: prompt, role: 'user' }
        ]
    };

    try {
        const response = await fetch(config.apiUrl, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
                'Authorization': `Bearer ${config.apiKey}`
            },
            body: JSON.stringify(requestData)
        });

        const reader = response.body.getReader();
        const decoder = new TextDecoder();
        
        let fullContent = '';
        
        while (true) {
            const { done, value } = await reader.read();
            if (done) break;

            const chunk = decoder.decode(value);
            const lines = chunk.split('\n');

            for (const line of lines) {
                if (line.startsWith('data: ')) {
                    const data = line.slice(6);
                    if (data === '[DONE]') continue;
                    
                    try {
                        const parsed = JSON.parse(data);
                        const delta = parsed.choices[0]?.delta?.content;
                        if (delta) {
                            fullContent += delta;
                            // 实时更新UI
                            updateAvatarContent(fullContent);
                        }
                    } catch (e) {
                        // 忽略解析错误
                    }
                }
            }
        }

        // 流式接收完成后，让数字人讲解
        await speakMessage(fullContent);
        
    } catch (error) {
        console.error('API调用失败:', error);
    }
}
```

### 文本分句优化

为了更好的讲解体验，我们对大模型的输出进行按照标点符号分句处理：

```javascript
function splitTextIntoSentences(text) {
    const sentences = text.match(/[^。！？!?]+[。！？!?]?/g) || [];
    return sentences.filter(s => s.trim());
}

// 逐句讲解
async function speakTextBySentences(text) {
    const sentences = splitTextIntoSentences(text);
    
    for (const sentence of sentences) {
        if (!isSpeaking) break;
        await speakMessage(sentence);
        // 等待当前句子讲完
        await waitForSpeakingComplete();
    }
}
```
